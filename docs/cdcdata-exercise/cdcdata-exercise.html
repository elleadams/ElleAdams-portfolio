The data for this exercise is the Weekly Respiratory Syncytial Virus (RSV) Vaccination Coverage among Pregnant Women by Race and Ethnicity age 18-49 in 2023-2024, gathered by National Center for Immunization and Respiratory Diseases (NCIRD), collected from the CDC’s data website.Link to data: https://data.cdc.gov/Pregnancy-Vaccination/Weekly-Respiratory-Syncytial-Virus-RSV-Vaccination/g4jn-64pd/about_data The seven columns include the week ending date, race and ethnicity, percent of pregnant women who were vaccinated, denominator being the estimated number of pregnant women, date order, race sort order, and figure ID.

I then modified values like the adding all of the race and ethnicity categories, modifying the number of weeks, the noise levels, and some of the logistic equations to make the graphical output more similar ot the original.

set.seed(123)

# Weekly dates
dates <- seq.Date(as.Date("2023-01-07"), by = "week", length.out = 27)

races <- c("White, NH", "Black, NH", "Asian, NH", 
           "Hispanic", "Multiple/Other", 
           "American Indian", 
           "Native Hawaiian/Pacific Islander", 
           "Unknown")

syn_dat <- do.call(rbind, lapply(races, function(r) {
  
  t <- 1:length(dates)
  
  percent <- switch(r,
    "White, NH"  = 0.18 / (1 + exp(-0.15 * (t - 20))),
    "Black, NH"  = 0.10 / (1 + exp(-0.10 * (t - 22))),
    "Asian, NH"  = 0.2 / (1 + exp(-0.20 * (t - 18))),  # fastest growth
    "Hispanic"   = 0.150 / (1 + exp(-0.13 * (t - 21))),
    "Multiple/Other" = 0.140 / (1 + exp(-0.12 * (t - 22))),
    "American Indian" = 0.120 / (1 + exp(-0.11 * (t - 23))),
    "Native Hawaiian/Pacific Islander" = 0.130 / (1 + exp(-0.14 * (t - 21))),
    "Unknown" = 0.100 / (1 + exp(-0.09 * (t - 24)))
  )
  
  # Add week-to-week noise
  percent <- percent + rnorm(length(percent), 0, 0.002)
  percent[percent < 0] <- 0
  percent[percent > 0.178] <- 0.178
  
  total_preg <- round(runif(length(dates), 8000, 20000))
  number_vax <- round(total_preg * percent)
  percent_of_total_vax = round(100 * number_vax / total_preg, 1)
  
  data.frame(
    week_ending_date = dates,
    race_and_ethnicity = r,
    percent_vaxed = percent,
    total_preg = total_preg,
    number_vax = number_vax,
    percent_of_total_vax = percent_of_total_vax
  )
}))
